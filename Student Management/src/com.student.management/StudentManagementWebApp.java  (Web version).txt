package com.student.management;

import com.sun.net.httpserver.HttpServer;
import java.io.IOException;
import java.net.InetSocketAddress;

/**
 * Student Management System - Web Server
 * Starts HTTP server for web-based student management application
 * 
 * Usage:
 * 1. Run this main class
 * 2. Open http://localhost:8080 in browser
 * 3. Use web interface for CRUD operations
 */
public class StudentManagementWebApp {
    
    // Server configuration
    private static final int PORT = 7070;
    private static final String CONTEXT_PATH = "/";
    
    public static void main(String[] args) {
        try {
            // Create HTTP server
            HttpServer server = HttpServer.create(
                new InetSocketAddress(PORT), 
                0
            );
            
            // Create context handler
            server.createContext(CONTEXT_PATH, new StudentHandler());
            
            // Set thread pool
            server.setExecutor(java.util.concurrent.Executors.newFixedThreadPool(10));
            
            // Start server
            server.start();
            
            // Display startup message
            displayStartupMessage();
            
        } catch (IOException e) {
            System.err.println("❌ Error starting server: " + e.getMessage());
            e.printStackTrace();
            System.exit(1);
        }
    }
